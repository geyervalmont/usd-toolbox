cmake_minimum_required(VERSION 3.20)
project(usd_toolbox_c_example C)
if(NOT TOOLBOX_LIBRARY)
  message(FATAL_ERROR "Pass -DTOOLBOX_LIBRARY=/absolute/path/to/library")
endif()
add_executable(document document.c)
if(EXISTS "${CMAKE_CURRENT_LIST_DIR}/../../usd_toolbox.h")
  target_include_directories(document PRIVATE ../..)
else()
  target_include_directories(document PRIVATE ../../crates/ffi/include)
endif()
target_link_libraries(document PRIVATE "${TOOLBOX_LIBRARY}")
# Rust's musl static library needs a musl-compatible unwinder. Some musl-gcc
# wrappers otherwise select the host glibc libgcc_eh, which cannot link.
get_filename_component(TOOLBOX_LIBRARY_DIRECTORY "${TOOLBOX_LIBRARY}" DIRECTORY)
if(NOT TOOLBOX_UNWIND_LIBRARY AND EXISTS "${TOOLBOX_LIBRARY_DIRECTORY}/libunwind.a")
  set(TOOLBOX_UNWIND_LIBRARY "${TOOLBOX_LIBRARY_DIRECTORY}/libunwind.a")
endif()
if(TOOLBOX_UNWIND_LIBRARY)
  if(NOT EXISTS "${TOOLBOX_UNWIND_LIBRARY}")
    message(FATAL_ERROR "TOOLBOX_UNWIND_LIBRARY does not exist: ${TOOLBOX_UNWIND_LIBRARY}")
  endif()
  target_link_libraries(document PRIVATE "${TOOLBOX_UNWIND_LIBRARY}")
endif()
if(UNIX)
  find_package(Threads REQUIRED)
  target_link_libraries(document PRIVATE Threads::Threads ${CMAKE_DL_LIBS} m)
endif()
# Assertions exercise calls even in Release builds.
if(MSVC)
  target_compile_options(document PRIVATE /UNDEBUG /utf-8)
else()
  target_compile_options(document PRIVATE -UNDEBUG -Wall -Wextra -Werror)
endif()
enable_testing()
add_test(NAME retained_document COMMAND document)
